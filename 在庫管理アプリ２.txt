---

### 🔍 画面が黒くなる主な原因と対策

1. **スマホやブラウザの「強制ダークモード」による影響**
* 背景が白（`#f8fafc`）に設定されている箇所が、ブラウザの自動ダークモード機能で黒く反転してしまっている可能性があります。


2. **スクリプトのエラーによる強制停止**
* データの読み込みや描画時にエラーが起きると、画面が白や黒のままで止まってしまうことがあります。

        function renderInventory() {
            populateCategoryDropdowns();
            const filterCat = document.getElementById('filter-category').value;
            const filterStatus = document.getElementById('filter-status').value;
            const tbody = document.getElementById('inventory-table-body');
            tbody.innerHTML = '';

            let filtered = products.filter(p => {
                if (filterCat && p.category !== filterCat) return false;
                if (filterStatus === 'zero' && p.stock !== 0) return false;
                if (filterStatus === 'low' && (p.stock > p.minStock || p.stock === 0)) return false;
                return true;
            });

            if (filtered.length === 0) {
                tbody.innerHTML = '<tr><td colspan="6" style="text-align:center; color:var(--text-sub);">該当する商品はありません</td></tr>';
                return;
            }

            filtered.forEach(p => {
                let statusBadge = '';
                if (p.stock === 0) {
                    statusBadge = '<span class="badge badge-zero">在庫ゼロ</span>';
                } else if (p.stock <= p.minStock) {
                    statusBadge = '<span class="badge badge-low">要発注</span>';
                } else {
                    statusBadge = '<span class="badge badge-ok">適正</span>';
                }

                tbody.innerHTML += `
                    <tr>
                        <td><strong>${escapeHtml(p.name)}</strong></td>
                        <td>${escapeHtml(p.category || '未分類')}</td>
                        <td style="font-size:15px; font-weight:bold;">${p.stock}</td>

                `;

        function saveProductModal() {
            const id = document.getElementById('edit-product-id').value;
            const name = document.getElementById('modal-product-name').value.trim();
            const category = document.getElementById('modal-product-category').value || '未分類';
            const stock = parseInt(document.getElementById('modal-product-stock').value) || 0;
            const minStock = parseInt(document.getElementById('modal-product-min-stock').value) || 0;

            if (!name) {
                alert('商品名を入力してください');
                return;
            }

            if (id) {
                const p = products.find(item => item.id == id);
                if (p) {
                    p.name = name;
                    p.category = category;
                    p.stock = stock;
                    p.minStock = minStock;
                }
            } else {
                const newP = { id: Date.now(), name, category, stock, minStock };
                products.push(newP);
            }

            saveData();
            closeProductModal();
            renderInventory();
        }

        function deleteProduct(id) {
            if (!confirm('この商品を削除してもよろしいですか？')) return;
            products = products.filter(p => p.id !== id);
            saveData();

        }

        function prepareInoutTab() {
            const select = document.getElementById('inout-product-id');
            select.innerHTML = '';

                select.innerHTML += `<option value="${p.id}">${escapeHtml(p.name)} (現在: ${p.stock})</option>`;

        function processStockChange() {
            const pId = parseInt(document.getElementById('inout-product-id').value);
            const type = document.getElementById('inout-type').value;
            const qty = parseInt(document.getElementById('inout-qty').value) || 0;
            const note = document.getElementById('inout-note').value.trim();

            if (!pId || qty <= 0) {
                alert('正しい商品と数量を指定してください');
                return;
            }

            const product = products.find(p => p.id === pId);
            if (!product) return;

            if (type === 'out' && product.stock < qty) {
                if (!confirm('現在の在庫数を超えて出荷しようとしています。続行しますか？')) return;
            }

            const changeQty = type === 'in' ? qty : -qty;
            product.stock += changeQty;

            const log = {
                id: Date.now(),
                date: new Date().toLocaleString(),
                productName: product.name,
                type: type === 'in' ? '入荷' : '出荷',
                qty: changeQty > 0 ? `+${changeQty}` : `${changeQty}`,
                afterStock: product.stock,
                note: note || '-'
            };
            historyLogs.unshift(log);

            saveData();
            document.getElementById('inout-note').value = '';
            prepareInoutTab();
            alert('在庫数を更新しました');
        }

        function renderHistoryTable() {

                        <td>${h.qty}</td>
                        <td>${h.afterStock}</td>
                        <td>${escapeHtml(h.note)}</td>
                    </tr>
                `;
            });
        }

                        <td><strong>${escapeHtml(p.name)}</strong></td>
                        <td>${escapeHtml(p.category || '未分類')}</td>
                        <td style="color:var(--danger); font-weight:bold;">${p.stock}</td>
                        <td>${p.minStock}</td>
                        <td><strong style="color:var(--primary);">${suggestQty > 0 ? suggestQty : 1}</strong></td>
                    </tr>
                `;
            });
        }

        function renderSettings() {
            const container = document.getElementById('category-list-tags');
            if (categories.length === 0) {
                container.innerHTML = '<span style="font-size:12px; color:var(--text-sub);">登録されているカテゴリはありません</span>';
                return;
            }
            container.innerHTML = categories.map(c => `
                <div class="category-tag">
                    ${escapeHtml(c)}
                    <button onclick="deleteCategory('${escapeHtml(c)}')" title="削除">✕</button>
                </div>
            `).join('');
        }

        function addCategory() {
            const val = document.getElementById('new-category-name').value.trim();
            if (!val) {
                alert('カテゴリ名を入力してください');
                return;
            }
            if (categories.includes(val)) {
                alert('すでに登録されているカテゴリ名です');
                return;
            }
            categories.push(val);
            saveData();
            document.getElementById('new-category-name').value = '';
            renderSettings();
            populateCategoryDropdowns();
        }

        function deleteCategory(cat) {
            if (!confirm(`カテゴリ「${cat}」を削除しますか？\n※このカテゴリが設定されていた商品は「未分類」に変更されます。`)) return;
            
            categories = categories.filter(c => c !== cat);
            
            products.forEach(p => {
                if (p.category === cat) {
                    p.category = '未分類';
                }
            });

            saveData();
            renderSettings();
            populateCategoryDropdowns();
            renderInventory();
        }

        function populateCategoryDropdowns() {
            const filterSel = document.getElementById('filter-category');
            const modalSel = document.getElementById('modal-product-category');
            
            if (filterSel) {
                const currentVal = filterSel.value;
                filterSel.innerHTML = '<option value="">すべて</option>';
                categories.forEach(c => {
                    filterSel.innerHTML += `<option value="${escapeHtml(c)}">${escapeHtml(c)}</option>`;
                });
                filterSel.innerHTML += '<option value="未分類">未分類</option>';
                filterSel.value = currentVal;
            }

            if (modalSel) {
                const currentVal = modalSel.value;
                modalSel.innerHTML = '';
                categories.forEach(c => {
                    modalSel.innerHTML += `<option value="${escapeHtml(c)}">${escapeHtml(c)}</option>`;
                });
                modalSel.innerHTML += '<option value="未分類">未分類</option>';
                modalSel.value = currentVal;
            }
        }

        function exportInventoryCSV() {
            let csv = "\uFEFF商品ID,商品名,カテゴリ,現在在庫数,最低在庫数,要発注フラグ\n";
            products.forEach(p => {
                const isNeedOrder = p.stock <= p.minStock ? "要発注" : "適正";
                csv += `"${p.id}","${p.name}","${p.category || '未分類'}",${p.stock},${p.minStock},"${isNeedOrder}"\n`;
            });
            downloadFile(csv, `在庫一覧_${getFormattedDate()}.csv`, 'text/csv');
        }

        function exportHistoryCSV() {
            let csv = "\uFEFF日時,商品名,区分,変動数,変動後在庫,備考\n";
            historyLogs.forEach(h => {
                csv += `"${h.date}","${h.productName}","${h.type}",${h.qty},${h.afterStock},"${h.note}"\n`;
            });
            downloadFile(csv, `入出庫履歴_${getFormattedDate()}.csv`, 'text/csv');
        }

        function exportBackupJSON() {
            const backupData = {
                categories,
                products,
                historyLogs,
                exportedAt: new Date().toISOString()
            };
            downloadFile(JSON.stringify(backupData, null, 2), `在庫データバックアップ_${getFormattedDate()}.json`, 'application/json');
        }

        function importBackupJSON(event) {
            const file = event.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(e) {
                try {
                    const data = JSON.parse(e.target.result);
                    if (data.products && data.categories) {
                        if (confirm('現在のデータを上書きして復元しますか？')) {
                            categories = data.categories || [];
                            products = data.products || [];
                            historyLogs = data.historyLogs || [];
                            saveData();
                            alert('データの復元が完了しました！');
                            location.reload();
                        }
                    } else {
                        alert('無効なバックアップファイル形式です');
                    }
                } catch (err) {
                    alert('ファイルの読み込みに失敗しました');
                }
            };
            reader.readAsText(file);
        }

        function saveData() {
            localStorage.setItem('stock_categories', JSON.stringify(categories));
            localStorage.setItem('stock_products', JSON.stringify(products));
            localStorage.setItem('stock_history', JSON.stringify(historyLogs));
        }

        function downloadFile(content, filename, contentType) {
            const blob = new Blob([content], { type: contentType });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = filename;
            a.click();
            URL.revokeObjectURL(url);
        }

        function getFormattedDate() {
            return new Date().toISOString().slice(0, 10);
        }

        function escapeHtml(str) {
            if (!str) return '';
            return String(str).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
        }
